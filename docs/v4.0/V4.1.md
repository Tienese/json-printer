## 4. Technical Specifications

### 4.1 Conjugation Mapping

Sudachi uses official Japanese grammar terms. Map to friendly names:

```java
// src/main/java/.../util/ConjugationMapper.java
public class ConjugationMapper {
    
    // Sudachi conjugationForm → Friendly name
    private static final Map<String, String> FORM_MAP = Map.ofEntries(
        // 終止形 (terminal form) - dictionary/plain form
        entry("終止形-一般", "plain"),
        entry("終止形-撥音便", "plain"),
        
        // 連用形 (continuative form) - masu stem, te-form base
        entry("連用形-一般", "masu-stem"),
        entry("連用形-促音便", "masu-stem"),
        entry("連用形-撥音便", "masu-stem"),
        
        // 未然形 (irrealis form) - nai-form base
        entry("未然形-一般", "nai-stem"),
        entry("未然形-サ", "nai-stem"),
        
        // 仮定形 (hypothetical form) - conditional
        entry("仮定形-一般", "conditional"),
        
        // 命令形 (imperative form)
        entry("命令形", "imperative"),
        
        // 意志推量形 (volitional form)
        entry("意志推量形", "volitional")
    );
    
    // Combined forms (surface + next token analysis)
    public static String detectCombinedForm(String surface, String nextSurface) {
        if (surface.endsWith("て") || surface.endsWith("で")) return "te-form";
        if (surface.endsWith("た") || surface.endsWith("だ")) return "ta-form";
        if (surface.endsWith("ない")) return "nai-form";
        if (nextSurface != null && nextSurface.equals("ます")) return "masu-form";
        return null;
    }
    
    public static String mapToFriendly(String sudachiForm) {
        return FORM_MAP.getOrDefault(sudachiForm, sudachiForm);
    }
    
    public static boolean matchesForm(String sudachiForm, String requiredForm) {
        String friendly = mapToFriendly(sudachiForm);
        return friendly.equals(requiredForm) || sudachiForm.contains(requiredForm);
    }
}
```

### 4.2 VERB_FORM Condition Implementation

```java
// In SentenceValidationService.java
private boolean evaluateCondition(RuleCondition cond, TokenResult token, 
                                   TokenResult nextToken) {
    switch (cond.getConditionType()) {
        // ... existing cases ...
        
        case "VERB_FORM":
            if (!token.posLevel1().equals("動詞")) return false;
            
            // Check Sudachi conjugationForm
            String sudachiForm = token.conjugationForm();
            String required = cond.getConditionValue();
            
            // Try direct match first
            if (ConjugationMapper.matchesForm(sudachiForm, required)) {
                return true;
            }
            
            // Try combined form detection (te-form, ta-form, etc.)
            String combined = ConjugationMapper.detectCombinedForm(
                token.surface(), 
                nextToken != null ? nextToken.surface() : null
            );
            return required.equals(combined);
            
        default:
            return true;
    }
}
```

### 4.3 Pattern Rule Validation Enhancement

```java
// In SentenceValidationService.java
private List<ViolationDTO> evaluatePatternRule(
        GrammarRuleV4 rule, 
        List<TokenResult> tokens) {
    
    List<ViolationDTO> violations = new ArrayList<>();
    String pattern = rule.getPattern();
    
    // Find pattern in token stream
    for (int i = 0; i < tokens.size(); i++) {
        TokenResult token = tokens.get(i);
        
        // Check if this token starts the pattern
        if (matchesPattern(tokens, i, pattern)) {
            // Found pattern - now check conditions
            List<RuleCondition> conditions = conditionRepo
                .findByRuleIdOrderByConditionOrder(rule.getId());
            
            for (RuleCondition cond : conditions) {
                if (cond.getTargetPosition().equals("BEFORE")) {
                    // Check verb form BEFORE the pattern
                    TokenResult beforeToken = i > 0 ? tokens.get(i - 1) : null;
                    TokenResult beforeBeforeToken = i > 1 ? tokens.get(i - 2) : null;
                    
                    if (beforeToken != null && 
                        !evaluateCondition(cond, beforeToken, token)) {
                        violations.add(createViolation(rule, cond, beforeToken));
                    }
                }
            }
        }
    }
    
    return violations;
}

private boolean matchesPattern(List<TokenResult> tokens, int start, String pattern) {
    // Simple pattern matching - concatenate surfaces and check
    StringBuilder sb = new StringBuilder();
    for (int i = start; i < Math.min(start + 5, tokens.size()); i++) {
        sb.append(tokens.get(i).surface());
        if (sb.toString().contains(pattern)) {
            return true;
        }
    }
    return false;
}
```

### 4.4 VERB_PARTICLE Rule Type

Handle edge case verbs that take unexpected particles:

```java
// In SentenceValidationService.java
private List<ViolationDTO> evaluateVerbParticleRule(
        GrammarRuleV4 rule, 
        List<TokenResult> tokens) {
    
    List<ViolationDTO> violations = new ArrayList<>();
    
    // Find the specific verb
    for (int i = 0; i < tokens.size(); i++) {
        TokenResult token = tokens.get(i);
        
        if (token.posLevel1().equals("動詞") && 
            token.baseForm().equals(rule.getPattern())) {  // pattern holds verb baseForm
            
            // Check what particle precedes a noun before this verb
            // e.g., "東京に住みます" - check that に is used, not で
            
            List<RuleCondition> conditions = conditionRepo
                .findByRuleIdOrderByConditionOrder(rule.getId());
            
            for (RuleCondition cond : conditions) {
                if (cond.getConditionType().equals("REQUIRES_PARTICLE")) {
                    String requiredParticle = cond.getConditionValue();
                    String actualParticle = findParticleBefore(tokens, i);
                    
                    if (actualParticle != null && !actualParticle.equals(requiredParticle)) {
                        violations.add(new ViolationDTO(
                            rule.getId(),
                            rule.getName(),
                            String.format("%s requires particle %s, not %s", 
                                token.baseForm(), requiredParticle, actualParticle),
                            token.startOffset(),
                            token.surface()
                        ));
                    }
                }
            }
        }
    }
    
    return violations;
}

private String findParticleBefore(List<TokenResult> tokens, int verbIndex) {
    // Walk backwards to find the particle
    for (int i = verbIndex - 1; i >= 0; i--) {
        TokenResult t = tokens.get(i);
        if (t.posLevel1().equals("助詞")) {
            return t.surface();
        }
        if (t.posLevel1().equals("名詞")) {
            continue; // Skip nouns, keep looking for particle
        }
        break; // Stop if we hit something else
    }
    return null;
}
```

### 4.5 DIRTY Status Workflow

```java
// In SentenceService.java
@Transactional
public SentenceDTO createSentence(CreateSentenceRequest request) {
    Sentence sentence = new Sentence();
    sentence.setText(request.text());
    sentence.setLessonId(request.lessonId());
    sentence.setExpectedValid(request.expectedValid());
    sentence.setNotes(request.notes());
    sentence.setSource(request.source() != null ? request.source() : "manual");
    
    // Tokenize and check for unknown vocab
    List<TokenResult> tokens = tokenizerService.tokenize(request.text());
    List<String> unknownWords = findUnknownVocab(tokens);
    
    if (!unknownWords.isEmpty()) {
        sentence.setValidationStatus("DIRTY");
        sentence.setValidationMessage("Unknown vocabulary: " + String.join(", ", unknownWords));
    } else {
        sentence.setValidationStatus("UNCHECKED");
    }
    
    sentence = sentenceRepository.save(sentence);
    
    // Auto-link known vocab
    autoLinkVocab(sentence.getId(), tokens);
    
    return toDTO(sentence);
}

private List<String> findUnknownVocab(List<TokenResult> tokens) {
    List<String> unknown = new ArrayList<>();
    
    for (TokenResult token : tokens) {
        // Skip particles, punctuation, etc.
        if (!isContentWord(token)) continue;
        
        // Check if vocab exists
        Optional<Vocab> vocab = vocabRepository.findByWord(token.surface());
        if (vocab.isEmpty()) {
            vocab = vocabRepository.findByWord(token.baseForm());
        }
        
        if (vocab.isEmpty()) {
            unknown.add(token.surface());
        }
    }
    
    return unknown;
}

private boolean isContentWord(TokenResult token) {
    String pos = token.posLevel1();
    return pos.equals("名詞") || pos.equals("動詞") || 
           pos.equals("形容詞") || pos.equals("副詞");
}
```

### 4.6 Auto-Resolve DIRTY on Vocab Add

```java
// In VocabService.java (or create VocabEventListener)
@Transactional
public VocabDTO createVocab(CreateVocabRequest request) {
    Vocab vocab = // ... create vocab ...
    vocab = vocabRepository.save(vocab);
    
    // Trigger DIRTY resolution
    resolveDirtySentences(vocab.getWord());
    
    return toDTO(vocab);
}

private void resolveDirtySentences(String newWord) {
    // Find DIRTY sentences that might contain this word
    List<Sentence> dirtySentences = sentenceRepository
        .findByValidationStatus("DIRTY");
    
    for (Sentence sentence : dirtySentences) {
        if (sentence.getText().contains(newWord)) {
            // Re-check unknown vocab
            List<TokenResult> tokens = tokenizerService.tokenize(sentence.getText());
            List<String> stillUnknown = findUnknownVocab(tokens);
            
            if (stillUnknown.isEmpty()) {
                sentence.setValidationStatus("UNCHECKED");
                sentence.setValidationMessage(null);
            } else {
                sentence.setValidationMessage("Unknown vocabulary: " + 
                    String.join(", ", stillUnknown));
            }
            
            sentenceRepository.save(sentence);
        }
    }
}
```

```java
// src/main/java/.../service/SeedDataService.java
@Service
public class SeedDataService {
    
    private final GrammarRuleV4Repository ruleRepository;
    private final RuleConditionRepository conditionRepository;
    private final ObjectMapper objectMapper;
    
    @PostConstruct
    public void loadSeedData() {
        if (ruleRepository.count() > 0) {
            log.info("Seed data already exists, skipping");
            return;
        }
        
        loadPatternRules();
        loadVerbParticleRules();
    }
    
    private void loadPatternRules() {
        try {
            InputStream is = getClass().getResourceAsStream("/data/pattern-rules-seed.json");
            List<RuleSeedData> seeds = objectMapper.readValue(is, 
                new TypeReference<List<RuleSeedData>>() {});
            
            for (RuleSeedData seed : seeds) {
                GrammarRuleV4 rule = createRule(seed);
                rule = ruleRepository.save(rule);
                
                for (ConditionSeedData cond : seed.conditions()) {
                    createCondition(rule.getId(), cond);
                }
            }
            
            log.info("Loaded {} pattern rules", seeds.size());
        } catch (Exception e) {
            log.error("Failed to load pattern rules seed data", e);
        }
    }
    
    private void loadVerbParticleRules() {
        // Similar to above
    }
}
```
