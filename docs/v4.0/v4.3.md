## 1. Executive Summary

### 1.1 Problem Statement
Grammar Coach V4.0/V4.1 backend is complete with:
- 5 entities (Sentence, SentenceVocabLink, GrammarRuleV4, RuleCondition, RuleTestSentence)
- Sudachi tokenizer integration
- Validation engine with conjugation support
- REST API endpoints

However, there is **no frontend UI** to interact with these features. The existing `worksheet-ui` directory contains the old worksheet functionality but no Grammar Coach pages.

### 1.2 Proposed Solution
Create three Grammar Coach pages with full CRUD functionality:
1. **Vocabulary Management** - Browse vocab, edit tags, view linked sentences
2. **Sentence Bank** - CRUD sentences, validation status display
3. **Grammar Rules** - Form-based rule editor with conditions and test sentences

Additionally:
- Rename `worksheet-ui` → `frontend` for clarity
- Verify all DB ↔ UI data flows work correctly
- Include V5.0 skeleton placeholders (non-functional)
- Print-first design (B&W compatible + print stylesheet)

// src/main/frontend/src/api/sentenceApi.ts

const BASE_URL = '/api/sentences';

export const sentenceApi = {
  // List with filters
  list: async (filters?: SentenceFilters): Promise<Sentence[]> => {
    const params = new URLSearchParams();
    if (filters?.lessonId) params.append('lessonId', String(filters.lessonId));
    if (filters?.status && filters.status !== 'ALL') params.append('status', filters.status);
    if (filters?.search) params.append('search', filters.search);
    
    const response = await fetch(`${BASE_URL}?${params}`);
    if (!response.ok) throw new Error('Failed to fetch sentences');
    return response.json();
  },

  // Get by ID
  get: async (id: number): Promise<Sentence> => {
    const response = await fetch(`${BASE_URL}/${id}`);
    if (!response.ok) throw new Error('Failed to fetch sentence');
    return response.json();
  },

  // Create
  create: async (data: SentenceCreateRequest): Promise<Sentence> => {
    const response = await fetch(BASE_URL, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(data),
    });
    if (!response.ok) throw new Error('Failed to create sentence');
    return response.json();
  },

  // Update
  update: async (id: number, data: SentenceUpdateRequest): Promise<Sentence> => {
    const response = await fetch(`${BASE_URL}/${id}`, {
      method: 'PUT',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(data),
    });
    if (!response.ok) throw new Error('Failed to update sentence');
    return response.json();
  },

  // Delete
  delete: async (id: number): Promise<void> => {
    const response = await fetch(`${BASE_URL}/${id}`, { method: 'DELETE' });
    if (!response.ok) throw new Error('Failed to delete sentence');
  },

  // Validate single
  validate: async (id: number): Promise<Sentence> => {
    const response = await fetch(`${BASE_URL}/${id}/validate`, { method: 'POST' });
    if (!response.ok) throw new Error('Failed to validate sentence');
    return response.json();
  },

  // Validate all
  validateAll: async (): Promise<{ validated: number; errors: number }> => {
    const response = await fetch(`${BASE_URL}/validate-all`, { method: 'POST' });
    if (!response.ok) throw new Error('Failed to validate sentences');
    return response.json();
  },

  // Get by vocab
  getByVocab: async (vocabId: number): Promise<Sentence[]> => {
    const response = await fetch(`${BASE_URL}?vocabId=${vocabId}`);
    if (!response.ok) throw new Error('Failed to fetch sentences');
    return response.json();
  },
};

const BASE_URL = '/api/grammar-rules';

export const grammarRuleApi = {
  // List with filters

    if (filters?.particle) params.append('particle', filters.particle);

    if (!response.ok) throw new Error('Failed to fetch rules');
    return response.json();
  },

  // Get by ID with conditions and tests
  get: async (id: number): Promise<GrammarRuleV4 & { conditions: RuleCondition[]; testSentences: RuleTestSentence[] }> => {
    const response = await fetch(`${BASE_URL}/${id}`);
    if (!response.ok) throw new Error('Failed to fetch rule');
    return response.json();
  },

  // Create rule
  create: async (data: Partial<GrammarRuleV4>): Promise<GrammarRuleV4> => {
    const response = await fetch(BASE_URL, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(data),
    });
    if (!response.ok) throw new Error('Failed to create rule');
    return response.json();
  },

  // Update rule
  update: async (id: number, data: Partial<GrammarRuleV4>): Promise<GrammarRuleV4> => {
    const response = await fetch(`${BASE_URL}/${id}`, {
      method: 'PUT',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(data),
    });
    if (!response.ok) throw new Error('Failed to update rule');
    return response.json();
  },

  // Delete rule
  delete: async (id: number): Promise<void> => {
    const response = await fetch(`${BASE_URL}/${id}`, { method: 'DELETE' });
